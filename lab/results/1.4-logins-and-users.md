### Switching users

```console label="Terminal"
$ sudo su - operator1 -c 'whoami; pwd; echo $HOME'
operator1
/home/operator1
/home/operator1
$ sudo su operator1 -c 'whoami; pwd; echo $HOME'
operator1
/home/student
/home/operator1
$ sudo -i -u operator1 whoami
operator1
$ sudo -i whoami
root
$ sudo grep -E '^(root|student|operator1):' /etc/passwd
root:x:0:0:Super User:/root:/bin/bash
student:x:1001:1001::/home/student:/bin/bash
operator1:x:1002:1002::/home/operator1:/bin/bash
```
