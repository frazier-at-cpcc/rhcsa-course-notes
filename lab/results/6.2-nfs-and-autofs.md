### autofs: an indirect map

```console label="Terminal"
$ rpm -q autofs
autofs-5.1.9-13.el10.x86_64
$ echo '/shares  /etc/auto.shares' | sudo tee /etc/auto.master.d/shares.autofs
/shares  /etc/auto.shares
$ echo 'projects  -rw,sync  servera:/exports/projects' | sudo tee /etc/auto.shares
projects  -rw,sync  servera:/exports/projects
$ sudo systemctl enable --now autofs
Created symlink '/etc/systemd/system/multi-user.target.wants/autofs.service' → '/usr/lib/systemd/system/autofs.service'.
$ ls /shares
$ ls /shares/projects
plan.txt
$ findmnt -t nfs4,nfs | grep shares
$ mount | grep -E 'autofs.*shares'
```

### autofs: a wildcard map for home directories

```console label="Terminal"
$ echo '/home/guests  /etc/auto.guests' | sudo tee /etc/auto.master.d/guests.autofs
/home/guests  /etc/auto.guests
$ echo '*  -rw,sync  servera:/exports/guests/&' | sudo tee /etc/auto.guests
*  -rw,sync  servera:/exports/guests/&
$ sudo systemctl reload autofs
$ cat /home/guests/alice/notes.txt
alice notes
$ ls /home/guests
alice
$ findmnt /home/guests/alice
TARGET             SOURCE                           FSTYPE OPTIONS
/home/guests/alice /dev/vda4[/exports/guests/alice] xfs    rw,relatime,seclabel,attr2,inode64,logbufs=8,logbsize=32k,noquota
$ grep -E '^(timeout|browse_mode)' /etc/autofs.conf
timeout = 300
browse_mode = no
```
