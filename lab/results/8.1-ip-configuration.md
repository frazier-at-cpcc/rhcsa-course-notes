### A static IPv4 and IPv6 connection

```console label="Terminal"
$ sudo nmcli connection add type ethernet con-name lab-static ifname ens19 ipv4.method manual ipv4.addresses 192.0.2.10/24 ipv4.dns 192.0.2.53 ipv6.method manual ipv6.addresses 2001:db8::10/64 connection.autoconnect yes

$ sudo nmcli connection up lab-static

$ ip -br addr show ens19

manual
192.0.2.10/24

yes
$ sudo cat /etc/NetworkManager/system-connections/lab-static.nmconnection
[connection]
id=lab-static
uuid=d08679b3-ef79-47bd-a113-60799debda65
type=ethernet
interface-name=ens19

[ethernet]

[ipv4]
address1=192.0.2.10/24
dns=192.0.2.53;
method=manual

[ipv6]
addr-gen-mode=default
address1=2001:db8::10/64
method=manual

[proxy]
```

```console label="Terminal"
$ sudo nmcli connection up lab-static; ip -br addr show ens19
```
